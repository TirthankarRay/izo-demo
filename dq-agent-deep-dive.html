⚙️ Agent State Machine

Internal state transitions that govern the DQ Agent's execution lifecycle
IDLE
Agent is waiting for next scheduled scan or incoming event from orchestration layer
→ on(tick | event) → PLANNING
PLANNING
Scanning registered data products, pulling SLA metadata, building priority queue
→ on(plan_ready) → EXECUTING
EXECUTING
Running validation checks across Monte Carlo, Snowflake, Enkrypt AI, Collibra
→ on(checks_done) → REFLECTING
REFLECTING
Correlating results, calculating blast radius, classifying severity
→ on(action_plan) → ACTING
ACTING
Executing auto-fixes, quarantining data, generating RCA, notifying teams
→ on(low_impact) → IDLE
HITL_WAITING
High-impact action detected — waiting for human approval before proceeding
→ on(approved) → ACTING
ERROR
Unrecoverable tool failure — paused with alert sent to Observability Agent
→ on(restart) → IDLE

from enum import Enum
from izo.agent import BaseAgent, AgentState, Transition

class DQAgentState(Enum):
    IDLE         = "idle"
    PLANNING     = "planning"
    EXECUTING    = "executing"
    REFLECTING   = "reflecting"
    ACTING       = "acting"
    HITL_WAITING = "hitl_waiting"
    ERROR        = "error"

class DataQualityAgent(BaseAgent):
    """iZO Data Quality Agent — continuous monitoring loop."""

    TRANSITIONS = [
        Transition(IDLE,         "tick",        PLANNING),
        Transition(IDLE,         "event",       PLANNING),
        Transition(PLANNING,     "plan_ready",  EXECUTING),
        Transition(EXECUTING,    "checks_done", REFLECTING),
        Transition(REFLECTING,   "action_plan", ACTING),
        Transition(ACTING,       "low_impact",  IDLE),          # auto-resolve
        Transition(ACTING,       "high_impact", HITL_WAITING),  # needs human
        Transition(HITL_WAITING, "approved",    ACTING),
        Transition(HITL_WAITING, "denied",      IDLE),
        Transition("*",           "error",       ERROR),
        Transition(ERROR,        "restart",     IDLE),
    ]

    def __init__(self, config: AgentConfig):
        super().__init__(
            agent_id="agent.dq",
            persona="Data Quality Engineer",
            schedule=config.scan_interval,  # default: every 5 minutes
            hitl_gate="ACT",
            escalation_threshold=config.escalation_threshold,
        )
        self.mc_client  = MonteCarloClient(config.monte_carlo)
        self.sf_client  = SnowflakeClient(config.snowflake)
        self.ek_client  = EnkryptAIClient(config.enkrypt)
        self.col_client = CollibraClient(config.collibra)
🧠 PHASE 1 — PLAN Assess Pipeline Health
Scan all registered data products. Prioritize by business impact score and SLA proximity.

The PLAN phase runs every 5 minutes (configurable). The agent queries Monte Carlo for active incidents, pulls freshness metadata from Snowflake's ACCOUNT_USAGE schema, and ranks all registered data assets by a composite score combining downstream consumer count, SLA proximity, and business impact tier.

Step-by-Step Execution

1

Query Monte Carlo for Active Incidents

Fetches all open incidents and anomaly alerts from Monte Carlo's REST API. Filters by severity (P1–P4) and maps each incident to its affected table/dataset in Snowflake.

2

Pull Freshness Metadata from Snowflake

Queries SNOWFLAKE.ACCOUNT_USAGE.TABLE_STORAGE_METRICS and ACCESS_HISTORY to determine when each table was last loaded. Compares against registered SLA thresholds to flag stale assets.

3

Rank Assets by Composite Impact Score

Computes impact_score = (downstream_consumers × 0.4) + (sla_urgency × 0.35) + (business_tier × 0.25). Business tier comes from Collibra's data product registry where each asset is tagged as Critical/High/Medium/Low.

Pythonagent_dq/phases/plan.py
async def plan(self, ctx: AgentContext) -> PlanResult:
    """Phase 1: Assess pipeline health and build priority queue."""

    # 1. Query Monte Carlo for active incidents
    incidents = await self.mc_client.get_incidents(
        status="open",
        severity=["P1", "P2", "P3"],
        created_after=ctx.last_scan_time,
    )
    ctx.memory.set("active_incidents", incidents)
    logger.info(f"Monte Carlo: {len(incidents)} active incidents found")

    # 2. Pull freshness metadata from Snowflake
    freshness = await self.sf_client.query("""
        SELECT table_catalog, table_schema, table_name,
               last_altered, bytes,
               DATEDIFF('minute', last_altered, CURRENT_TIMESTAMP()) AS stale_minutes
        FROM   SNOWFLAKE.ACCOUNT_USAGE.TABLE_STORAGE_METRICS
        WHERE  table_catalog = 'NOVARTIS_DW'
          AND  deleted IS NULL
        ORDER BY stale_minutes DESC
    """)

    # 3. Cross-reference with SLA registry
    sla_registry = await self.col_client.get_catalog(
        domain="novartis-pharma",
        asset_type="DataProduct",
        fields=["name", "sla_minutes", "business_tier", "consumers"],
    )

    # 4. Compute composite impact score
    priority_queue = []
    for asset in sla_registry:
        consumers     = len(asset["consumers"])
        stale_entry   = freshness.get(asset["name"])
        sla_urgency   = min(1.0, stale_entry.stale_minutes / asset["sla_minutes"]) if stale_entry else 0
        business_tier = {"Critical":1.0, "High":0.75, "Medium":0.5, "Low":0.25}[asset["business_tier"]]

        score = (consumers * 0.40) + (sla_urgency * 0.35) + (business_tier * 0.25)
        priority_queue.append(PriorityItem(asset=asset, score=score, incidents=incidents.get(asset["name"], [])))

    priority_queue.sort(key=lambda x: x.score, reverse=True)
    ctx.memory.set("priority_queue", priority_queue[:50])  # Top 50 per cycle

    return PlanResult(
        assets_scanned=len(sla_registry),
        incidents_found=len(incidents),
        queue_depth=len(priority_queue),
        top_asset=priority_queue[0].asset["name"] if priority_queue else None,
    )
🔧 PHASE 2 — TOOL USE Execute Quality Checks
Run validation rules across all data layers using native platform APIs.

For each asset in the priority queue, the agent dispatches tool-specific checks in parallel. It uses Monte Carlo's ML-based anomaly detection, Snowflake's built-in Data Metric Functions (DMFs), Enkrypt AI for PHI/PII scanning, and Collibra for lineage validation. All calls run concurrently via asyncio.gather.

async def tool_use(self, ctx: AgentContext) -> ToolUseResult:
    """Phase 2: Execute quality checks in parallel across all tools."""

    queue = ctx.memory.get("priority_queue")
    all_findings = []

    for asset in queue:
        # Run all 4 tool checks concurrently
        mc_task  = self._check_monte_carlo(asset)
        sf_task  = self._check_snowflake_dmf(asset)
        ek_task  = self._check_enkrypt_ai(asset)
        col_task = self._check_collibra_lineage(asset)

        results = await asyncio.gather(
            mc_task, sf_task, ek_task, col_task,
            return_exceptions=True,  # don't fail-fast
        )

        # Aggregate findings per asset
        findings = FindingSet(asset=asset)
        for result in results:
            if isinstance(result, Exception):
                findings.add_error(tool=result.tool, error=str(result))
                ctx.emit("tool_failure", tool=result.tool, asset=asset.name)
            else:
                findings.merge(result)

        all_findings.append(findings)

    ctx.memory.set("findings", all_findings)
    return ToolUseResult(
        assets_checked=len(queue),
        total_findings=sum(f.count for f in all_findings),
        tool_failures=sum(1 for f in all_findings if f.has_errors),
    )
💭 PHASE 3 — REFLECT Analyze Results & Impact
Correlate findings across tools. Determine root cause and blast radius. Classify: auto-resolve vs. escalate.

The Reflect phase is the "brain" of the agent. It uses Snowflake Cortex's CORTEX.COMPLETE() LLM function to correlate findings from different tools, identify root causes, calculate the blast radius (which downstream data products are impacted), and make the critical decision: can this be auto-resolved, or does it need human intervention?

Pythonagent_dq/phases/reflect.py
async def reflect(self, ctx: AgentContext) -> ReflectResult:
    """Phase 3: Correlate, classify severity, decide action plan."""

    findings = ctx.memory.get("findings")
    action_items = []

    for finding_set in findings:
        asset = finding_set.asset

        # 1. Cross-reference Monte Carlo alerts with Snowflake query logs
        correlated = await self._correlate_with_query_history(
            mc_anomalies=finding_set.mc_anomalies,
            sf_query_log=await self.sf_client.query(f"""
                SELECT query_text, error_message, execution_status,
                       start_time, warehouse_name
                FROM   SNOWFLAKE.ACCOUNT_USAGE.QUERY_HISTORY
                WHERE  query_text ILIKE '%{asset.table_name}%'
                  AND  start_time > DATEADD('hour', -24, CURRENT_TIMESTAMP())
                  AND  execution_status = 'FAIL'
                ORDER BY start_time DESC LIMIT 50
            """)
        )

        # 2. Map blast radius via Collibra lineage
        downstream = await self.col_client.get_lineage(
            asset_id=asset.collibra_id,
            direction="DOWNSTREAM",
            depth=3,
        )
        blast_radius = BlastRadius(
            affected_products=downstream.data_products,
            affected_kpis=downstream.kpi_definitions,
            affected_consumers=downstream.consumer_teams,
        )

        # 3. Use Cortex LLM for root cause analysis
        rca_prompt = self._build_rca_prompt(finding_set, correlated, blast_radius)
        rca = await self.sf_client.cortex_complete(
            model="llama3.1-70b",
            prompt=rca_prompt,
            max_tokens=500,
        )

        # 4. Classify: auto-resolve vs. escalate to human
        severity = self._classify_severity(finding_set, blast_radius)
        #   LOW    → auto-resolve (quarantine + fix)
        #   MEDIUM → auto-resolve + notify team
        #   HIGH   → requires HITL approval before acting
        #   CRIT   → immediate HITL escalation + page on-call

        action_items.append(ActionItem(
            asset=asset,
            severity=severity,
            root_cause=rca,
            blast_radius=blast_radius,
            requires_hitl=severity in ("HIGH", "CRITICAL"),
            remediation=self._select_remediation(finding_set, severity),
        ))

    ctx.memory.set("action_items", action_items)
    return ReflectResult(
        total_items=len(action_items),
        hitl_required=sum(1 for a in action_items if a.requires_hitl),
        auto_resolvable=sum(1 for a in action_items if not a.requires_hitl),
    )

Pythonagent_dq/phases/act.py
async def act(self, ctx: AgentContext) -> ActResult:
    """Phase 4: Execute remediation with HITL gating for high-impact."""

    action_items = ctx.memory.get("action_items")
    results = []

    for item in action_items:

        # ─── HITL GATE: Check if human approval is required ───
        if item.requires_hitl:
            approval = await self.request_hitl_approval(
                action=item.remediation.description,
                severity=item.severity,
                blast_radius=item.blast_radius.summary(),
                root_cause=item.root_cause,
                reviewer=self._get_reviewer(item.severity),
                timeout_minutes=30,  # auto-escalate after 30min
            )

            if approval.status == "denied":
                results.append(ActionResult(item, status="blocked", reason=approval.comment))
                continue
            elif approval.status == "timeout":
                # Auto-escalate to VP
                await self.escalate(item, to="vp_data", reason="HITL timeout")
                continue

        # ─── Execute Remediation Actions ───
        match item.remediation.action_type:

            case "quarantine":
                # Move affected records to quarantine schema
                await self.sf_client.query(f"""
                    INSERT INTO NOVARTIS_DW.QUARANTINE.{item.asset.table_name}
                    SELECT *, CURRENT_TIMESTAMP() AS quarantined_at,
                           '{item.root_cause.summary}' AS reason
                    FROM   NOVARTIS_DW.STAGING.{item.asset.table_name}
                    WHERE  {item.remediation.filter_clause}
                """)
                # Delete quarantined records from staging
                await self.sf_client.query(f"""
                    DELETE FROM NOVARTIS_DW.STAGING.{item.asset.table_name}
                    WHERE  {item.remediation.filter_clause}
                """)

            case "re_ingest":
                # Trigger Snowpark re-ingestion pipeline
                await self.sf_client.snowpark_exec(
                    procedure="NOVARTIS_DW.PIPELINES.REINGEST_TABLE",
                    args=[item.asset.table_name, item.remediation.source_system],
                )

            case "schema_fix":
                # Apply schema migration via ALTER TABLE
                for migration in item.remediation.schema_changes:
                    await self.sf_client.query(migration.sql)

        # ─── Generate RCA Report via Cortex LLM ───
        rca_report = await self.sf_client.cortex_complete(
            model="llama3.1-70b",
            prompt=f"""Generate a plain-English root cause analysis:
            Asset: {item.asset.name}
            Issue: {item.root_cause.summary}
            Blast radius: {item.blast_radius.summary()}
            Action taken: {item.remediation.description}
            Format: 3 bullet points for leadership, 1 paragraph for engineers."""
        )

        # ─── Notify via Slack ───
        await self.slack_client.send_alert(
            channel="#novartis-data-quality",
            severity=item.severity,
            title=f"DQ Agent: {item.remediation.description}",
            body=rca_report,
            fields={
                "Asset": item.asset.name,
                "Records Affected": item.blast_radius.record_count,
                "Downstream KPIs": ", ".join(item.blast_radius.affected_kpis),
                "HITL": "✅ Approved" if item.requires_hitl else "Auto-resolved",
            },
        )

        results.append(ActionResult(item, status="completed"))

    ctx.emit("cycle_complete", results=results)
    return ActResult(
        completed=sum(1 for r in results if r.status == "completed"),
        blocked=sum(1 for r in results if r.status == "blocked"),
    )

👤 Human-in-the-Loop Gate Logic

How the DQ Agent decides when to pause and request human oversight
Pythonagent_dq/hitl/gate.py
class HITLGate:
    """Human-in-the-Loop gate for the DQ Agent's ACT phase."""

    # Rules that trigger HITL — if ANY rule matches, human approval is required
    RULES = [
        HITLRule(
            name="mass_quarantine",
            condition=lambda item: item.blast_radius.record_count > 5000,
            reviewer="data_steward",
            reason="Mass data quarantine affects >5,000 records",
        ),
        HITLRule(
            name="schema_propagation",
            condition=lambda item: item.remediation.action_type == "schema_fix",
            reviewer="data_architect",
            reason="Schema changes require architect approval",
        ),
        HITLRule(
            name="cross_system_reingest",
            condition=lambda item: item.remediation.action_type == "re_ingest"
                       and item.remediation.source_system in ("veeva", "iqvia"),
            reviewer="data_steward",
            reason="Re-ingestion from external system requires validation",
        ),
        HITLRule(
            name="kpi_impact",
            condition=lambda item: len(item.blast_radius.affected_kpis) > 2,
            reviewer="vp_analytics",
            reason="Action impacts >2 downstream KPI definitions",
        ),
        HITLRule(
            name="regulatory_flag",
            condition=lambda item: item.asset.business_tier == "Critical"
                       and item.severity == "CRITICAL",
            reviewer="compliance_officer",
            reason="Critical-tier asset with regulatory implications",
        ),
    ]

    async def evaluate(self, item: ActionItem) -> HITLDecision:
        """Evaluate whether this action requires human approval."""
        triggered_rules = [r for r in self.RULES if r.condition(item)]

        if not triggered_rules:
            return HITLDecision(required=False, auto_approve=True)

        # Pick highest-authority reviewer if multiple rules triggered
        reviewer = self._highest_authority(
            [r.reviewer for r in triggered_rules]
        )

        return HITLDecision(
            required=True,
            rules_triggered=[r.name for r in triggered_rules],
            reasons=[r.reason for r in triggered_rules],
            reviewer=reviewer,
            timeout_minutes=30,       # escalate if no response
            escalation_chain=["vp_data", "cdo"],
        )
